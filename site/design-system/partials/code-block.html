psql → Provisa, port 5439
# psql speaks to Provisa as if it were Postgres
psql -h localhost -p 5439 -U analyst

# One query, joined across Postgres, Mongo and Elasticsearch
SELECT c.name, o.total, t.trace_id
FROM customers c
JOIN orders o   ON o.customer_id = c.id
JOIN queries t  ON t.actor = c.email
WHERE o.total > 1000;
provisa serve --config provisa.yaml
